To see the password expiration date -

Get-ADUser -Identity "user_id" -Properties "msDS-UserPasswordExpiryTimeComputed" | 
Select-Object Name, @{Name="PasswordExpiryDate";Expression={[datetime]::FromFileTime($_."msDS-UserPasswordExpiryTimeComputed")}}

Get-ADUser -Identity "user_id" -Properties AccountExpirationDate | 
Select-Object Name, SamAccountName, AccountExpirationDate
